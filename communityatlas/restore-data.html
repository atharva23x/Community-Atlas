Move CommunityAtlas data to localhost

Copies your reports, accounts, votes, and comments from the deployed site to 127.0.0.1:5500. Everything stays in this browser; the file is only a transfer container.

Step 1 - Copy your data from the deployed site

Open communityatlas-vert.vercel.app, then open DevTools (F12) and paste this into the Console. It copies all CommunityAtlas keys to your clipboard.

(() => {
  const data = {};
  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    if (key && key.startsWith("communityatlas")) {
      data[key] = localStorage.getItem(key);
    }
  }
  copy(JSON.stringify({
    app: "CommunityAtlas",
    exportedAt: new Date().toISOString(),
    origin: location.origin,
    data
  }));
  console.log("Copied " + Object.keys(data).length + " CommunityAtlas key(s). Now open the localhost restore page.");
})();
Prefer a file? Show the download snippet 
(() => {
  const data = {};
  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    if (key && key.startsWith("communityatlas")) {
      data[key] = localStorage.getItem(key);
    }
  }
  const blob = new Blob([JSON.stringify({
    app: "CommunityAtlas",
    exportedAt: new Date().toISOString(),
    origin: location.origin,
    data
  }, null, 2)], { type: "application/json" });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob);
  a.download = "communityatlas-data.json";
  a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 1000);
})();
If Chrome blocks pasting, type allow pasting in the console first.
